function circleCenter3D(P1, P2, r) {
  // Calculate the midpoint (M) of the chord.
  M = (P1 + P2) / 2; // Vector addition and scalar division

  // Calculate the chord vector (V).
  V = P2 - P1; // Vector subtraction

  // Define the "up" vector (assuming general orientation).
  up = [0, 1, 0];

  // Calculate the plane normal vector (planeNormal).
  planeNormal = normalize(crossProduct(V, up));

  // Calculate the perpendicular vector (N) from M to the center.
  N = normalize(crossProduct(planeNormal, V));

  // Calculate the chord length (c).
  c = magnitude(V);

  // Calculate the distance (d) from M to the circle center.
  d = sqrt(r^2 - (c/2)^2);

  // Calculate the circle center (C).
  C = M + N * d; // Vector addition and scalar multiplication

  return C;
}
